message(STATUS "Configure CMakeCompileFrameinterpolationShaders.txt")
message(STATUS "FFX_GLSLANG_EXE_PATH=${FFX_GLSLANG_EXE_PATH}")


set(ARM_FRAMEINTERPOLATION_BASE_ARGS
    -glslangexe=${FFX_GLSLANG_EXE_PATH}
    -DOUTPUT_IMG_FORMAT=r11f_g11f_b10f
    -reflection -deps=gcc -DFFX_GPU=1 -DFFX_8_BIT_TYPES=1)

if(WIN32)
    set(ARM_FRAMEINTERPOLATION_PERMUTATION_ARGS
        -DFFX_ARM_FRAMEINTERPOLATION_OPTION_LOW_RES_MOTION_VECTORS={0,1}
        -DFFX_ARM_FRAMEINTERPOLATION_OPTION_JITTER_MOTION_VECTORS={0,1}
        -DFFX_ARM_FRAMEINTERPOLATION_OPTION_INVERTED_DEPTH={0,1}
        -DFFX_ARM_FRAMEINTERPOLATION_OPTION_ENABLE_DATA_GRAPH_FI={0,1}
        -DFFX_ARM_FRAMEINTERPOLATION_OPTION_SUPPORT_TENSOR={0,1}
        -DFFX_ARM_FRAMEINTERPOLATION_OPTION_MANAGE_PREVIOUS_DEPTH={0,1}
        -DFFX_ARM_FRAMEINTERPOLATION_OPTION_MANAGE_PREVIOUS_COLOR={0,1})
else()
    set(ARM_FRAMEINTERPOLATION_PERMUTATION_ARGS
        -DFFX_ARM_FRAMEINTERPOLATION_OPTION_LOW_RES_MOTION_VECTORS="{0,1}"
        -DFFX_ARM_FRAMEINTERPOLATION_OPTION_JITTER_MOTION_VECTORS="{0,1}"
        -DFFX_ARM_FRAMEINTERPOLATION_OPTION_INVERTED_DEPTH="{0,1}"
        -DFFX_ARM_FRAMEINTERPOLATION_OPTION_ENABLE_DATA_GRAPH_FI="{0,1}"
        -DFFX_ARM_FRAMEINTERPOLATION_OPTION_SUPPORT_TENSOR="{0,1}"
        -DFFX_ARM_FRAMEINTERPOLATION_OPTION_MANAGE_PREVIOUS_DEPTH="{0,1}"
        -DFFX_ARM_FRAMEINTERPOLATION_OPTION_MANAGE_PREVIOUS_COLOR="{0,1}")
endif()

set(ARM_FRAMEINTERPOLATION_INCLUDE_ARGS
	"${FFX_GPU_PATH}"
	"${FFX_GPU_PATH}/frameinterpolation")

set(ARM_FRAMEINTERPOLATION_API_COMMON_ARGS
    -compiler=glslang -e main --target-env vulkan1.1 -Os -DFFX_GLSL=1)

file(GLOB ARM_FRAMEINTERPOLATION_SHADERS_COMP
    "shaders/frameinterpolation/*.glsl"
)
list(FILTER ARM_FRAMEINTERPOLATION_SHADERS_COMP EXCLUDE REGEX ".*_fs\\.glsl$")
list(FILTER ARM_FRAMEINTERPOLATION_SHADERS_COMP EXCLUDE REGEX ".*_vs\\.glsl$")

file(GLOB ARM_FRAMEINTERPOLATION_SHADERS_FRAG
    "shaders/frameinterpolation/*_fs.glsl"
)

file(GLOB ARM_FRAMEINTERPOLATION_SHADERS_VERT
    "shaders/frameinterpolation/*_vs.glsl"
)

compile_all_shaders(
    "${FFX_SC_EXECUTABLE}"
    "${ARM_FRAMEINTERPOLATION_BASE_ARGS}" "${ARM_FRAMEINTERPOLATION_API_COMMON_ARGS}"
    "${ARM_FRAMEINTERPOLATION_PERMUTATION_ARGS}" "${ARM_FRAMEINTERPOLATION_INCLUDE_ARGS}"
    "${ARM_FRAMEINTERPOLATION_SHADERS_COMP}" "${ARM_FRAMEINTERPOLATION_SHADERS_FRAG}" "${ARM_FRAMEINTERPOLATION_SHADERS_VERT}"
    "${FFX_PASS_SHADER_OUTPUT_PATH}" ARM_FRAMEINTERPOLATION_PERMUTATION_OUTPUTS)

add_shader_output("${ARM_FRAMEINTERPOLATION_PERMUTATION_OUTPUTS}")

# Data graphs: NO LONGER GENERATED.
#
# This used to call compile_data_graphs() with VGF_PARSER_EXECUTABLE
# (sdk/tools/binary_store/Model_Parser.exe) to turn `data_graphs/nfru/*.vgf` into
# nfru_v1_int8.h.
#
# Both halves of that are gone. The serializer crashes (0xC0000409) on this
# repository's own .vgf files, and it is no longer needed anyway: the header it
# produced was consumed only by CreateDataGraphPipelineVK, which fed the VGF bytes
# to vkCreateDataGraphPipelinesARM. That ARM execution path is replaced by a
# portable dp4a compute backend which carries its own baked int8 weights and never
# reads a VGF.
#
# The descriptor is checked in, hand-written from authoritative sources, at
# `sdk/src/backends/vk/data_graphs/nfru_v1_int8.h`.
# -----------------------------------------------------------------------------
